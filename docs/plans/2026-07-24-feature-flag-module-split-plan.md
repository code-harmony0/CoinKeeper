# Feature-Flag / Module-Split Foundation Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Make the codebase produce two build variants — `public` (Finance Tracker only) and `private` (Finance Tracker + Chit Fund) — driven by a single build-time env var, with a central module registry gating chit-fund navigation.

**Architecture:** `react-native-config` exposes `APP_VARIANT` (set via `.env.public` / `.env.private`, copied to `.env` before build) to JS. `src/config/modules.ts` derives `ENABLED_MODULES` from that variant. `RootNavigator` and `UserTabNavigator` read `isModuleEnabled('chitFund')` to decide which screens/tabs to register. See `docs/plans/2026-07-24-feature-flag-module-split-design.md` for the full module split rationale.

**Tech Stack:** React Native 0.80, TypeScript, `react-native-config`, Jest.

import Config from 'react-native-config';

**Step 3: Refactor UserTabNavigator.tsx to export a pure function and use the module registry**

```typescript
import React from 'react';
import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
import { UserDashboardScreen, TransactionsScreen, SettingsScreen } from '../screens/user';
import { ExpandingTabBar } from '../components/shared/ExpandingTabBar';
import { AddScreen } from '../screens/user/AddScreen';
import { KazaScreen } from '../screens/user/KazaScreen';
import { isModuleEnabled } from '../config/modules';

const Tab = createBottomTabNavigator();

type TabDef = { name: string; label: string; icon: 'home' | 'transactions' | 'add' | 'kaza' | 'settings' };

export const getUserTabs = (chitFundEnabled: boolean): TabDef[] => {
  const tabs: TabDef[] = [
    { name: 'Home', label: 'Home', icon: 'home' },
    { name: 'Transactions', label: 'Activity', icon: 'transactions' },
    { name: 'Add', label: 'Add', icon: 'add' },
  ];
  if (chitFundEnabled) {
    tabs.push({ name: 'Kaza', label: 'Kaza', icon: 'kaza' });
  }
  tabs.push({ name: 'Settings', label: 'Settings', icon: 'settings' });
  return tabs;
};

export const UserTabNavigator: React.FC = () => {
  const chitFundEnabled = isModuleEnabled('chitFund');
  const tabs = getUserTabs(chitFundEnabled);

  return (
    <Tab.Navigator
      tabBar={(props) => (
        <ExpandingTabBar {...props} tabs={tabs} hasFab fabIndex={2} />
      )}
      screenOptions={{ headerShown: false }}
    >
      <Tab.Screen name="Home" component={UserDashboardScreen} />
      <Tab.Screen name="Transactions" component={TransactionsScreen} />
      <Tab.Screen name="Add" component={AddScreen} />
      {chitFundEnabled && <Tab.Screen name="Kaza" component={KazaScreen} />}
      <Tab.Screen name="Settings" component={SettingsScreen} />
    </Tab.Navigator>
  );
};

export default UserTabNavigator;
```
